Edge Function: 

-- =========================
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
);

// List of architectural styles
const styles = ['Art Deco', 'Beaux-Arts', 'Gothic Revival', 'Modernist', 'Brutalist', 'Renaissance', 'Victorian'];

// Helper functions
const randomInt = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min;
const randomChoice = (array) => array[Math.floor(Math.random() * array.length)];

function weightedRandom(templates) {
  const totalWeight = templates.reduce((sum, t) => sum + t.weight, 0);
  let r = Math.random() * totalWeight;
  for (const t of templates) {
    if (r < t.weight) return t;
    r -= t.weight;
  }
  return templates[0];
}

function fillTemplate(template, variables) {
  return template.replace(/\{(\w+)\}/g, (_, key) => variables[key] || '');
}

// ======================
// Soft Validation
// ======================
function validateGPS(gpsPath, boundingBox) {
  if (!gpsPath || gpsPath.length === 0) return false;
  return gpsPath.some(point =>
    point.lat >= boundingBox.minLat &&
    point.lat <= boundingBox.maxLat &&
    point.lng >= boundingBox.minLng &&
    point.lng <= boundingBox.maxLng
  );
}

function validatePhoto(photoGPS, boundingBox) {
  if (!photoGPS) return false;
  return (
    photoGPS.lat >= boundingBox.minLat &&
    photoGPS.lat <= boundingBox.maxLat &&
    photoGPS.lng >= boundingBox.minLng &&
    photoGPS.lng <= boundingBox.maxLng
  );
}

function getBoundingBoxForQuest(quest) {
  // Example: simple bounding box around city or river area
  // Customize per quest.category / type
  return {
    minLat: 40.70, maxLat: 40.80,
    minLng: -74.02, maxLng: -73.95
  };
}

// ======================
// Generate Quest for a User
// ======================
async function generateQuestForUser(userId, type) {
  // 1️⃣ Get templates not used by this user in cooldown period
  const cooldownDays = type === 'daily' ? 7 : 14;
  const { data: templates, error } = await supabase
    .from('quest_templates')
    .select('*')
    .not('id', 'in', supabase
      .from('user_quests')
      .select('quest_template_id')
      .eq('user_id', userId)
      .gte('assigned_at', `now() - interval '${cooldownDays} days'`)
    );

  if (error || !templates || templates.length === 0) return;

  // 2️⃣ Pick weighted random template
  const template = weightedRandom(templates);

  // 3️⃣ Fill placeholders
  const count = randomInt(template.target_min, template.target_max);
  const style = randomChoice(styles);

  const quest = {
    user_id: userId,
    title: fillTemplate(template.title_template, { count, style }),
    description: fillTemplate(template.description_template, { count, style }),
    target: count,
    xp_reward: randomInt(template.xp_reward_min, template.xp_reward_max),
    stamp_template_id: template.stamp_template_id,
    active_until: type === 'daily'
      ? new Date(Date.now() + 24*60*60*1000)
      : new Date(Date.now() + 7*24*60*60*1000)
  };

  // 4️⃣ Insert quest
  const { error: insertError, data: insertedQuest } = await supabase
    .from('quests')
    .insert([quest])
    .select()
    .single();

  if (insertError) return console.error('Quest insert error:', insertError);

  // 5️⃣ Record assignment
  await supabase.from('user_quests').insert({
    user_id: userId,
    quest_template_id: template.id,
    assigned_at: new Date()
  });

  console.log(`Generated ${type} quest for user ${userId}: ${quest.title}`);
  return insertedQuest;
}

// ======================
// Validate Submission
// ======================
export async function validateQuestSubmission(submissionId) {
  const { data: submission, error } = await supabase
    .from('quest_submissions')
    .select('*, quests(type, category, xp_reward)')
    .eq('id', submissionId)
    .single();

  if (error || !submission) throw error || new Error('Submission not found');

  const boundingBox = getBoundingBoxForQuest(submission.quests);
  let isValid = false;

  if (['walk', 'scan'].includes(submission.quests.category)) {
    isValid = validateGPS(submission.gps_path, boundingBox);
  } else if (['style', 'find'].includes(submission.quests.category)) {
    isValid = validatePhoto(submission.photo_gps, boundingBox);
  } else {
    isValid = true; // creative / note quests
  }

  const xp = isValid ? submission.quests.xp_reward : 0;

  await supabase
    .from('quest_submissions')
    .update({ validated: isValid, xp_awarded: xp })
    .eq('id', submissionId);

  return { validated: isValid, xpAwarded: xp };
}

// ======================
// Main Runner
// ======================
export async function run() {
  const { data: users, error } = await supabase.from('users').select('id');
  if (error || !users) return console.error('User fetch error', error);

  const today = new Date();
  for (const user of users) {
   	await generateQuestForUser(user.id, 'daily');


    // Weekly quests only on Monday
    if (today.getDay() === 1) {
  	await generateQuestForUser(user.id, 'weekly');
    }
  }
}
